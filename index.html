function doGet(e) { var action = e.parameter.action; if (action === "getTeachers") { var sheet = SpreadsheetApp.getActive().getSheetByName("Teachers"); var data = sheet.getDataRange().getValues(); var teachers = []; for (var i = 1; i < data.length; i++) { if (data[i][0]) { teachers.push({ code: String(data[i][0]).trim(), name: String(data[i][1] || "").trim() }); } } return ContentService.createTextOutput(JSON.stringify({success:true, teachers:teachers})) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({success:true})) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var body = JSON.parse(e.postData.contents); var action = body.action; if (action === "register") { var sheet = SpreadsheetApp.getActive().getSheetByName("Teachers"); var data = sheet.getDataRange().getValues(); var code = String(body.teacherCode).trim(); var device = String(body.deviceId).trim(); for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === code) { if (String(data[i][2] || "").trim() !== "") { return ContentService.createTextOutput(JSON.stringify({success:false, message:"Already registered"})) .setMimeType(ContentService.MimeType.JSON); } sheet.getRange(i+1, 3).setValue(device); sheet.getRange(i+1, 4).setValue(new Date()); return ContentService.createTextOutput(JSON.stringify({success:true, message:"Registered"})) .setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({success:false, message:"Teacher not found"})) .setMimeType(ContentService.MimeType.JSON); } if (action === "markAttendance") { var type = String(body.type).toUpperCase(); var device = String(body.deviceId).trim(); var lat = Number(body.lat); var lng = Number(body.lng); // Find teacher var teacherSheet = SpreadsheetApp.getActive().getSheetByName("Teachers"); var teacherData = teacherSheet.getDataRange().getValues(); var teacher = null; for (var i = 1; i < teacherData.length; i++) { if (String(teacherData[i][2] || "").trim() === device) { teacher = { code: String(teacherData[i][0]).trim(), name: String(teacherData[i][1]).trim() }; break; } } if (!teacher) { return ContentService.createTextOutput(JSON.stringify({success:false, message:"Not registered"})) .setMimeType(ContentService.MimeType.JSON); } // Check distance var dist = getDistance(lat, lng, 26.8580694, 85.9583127); if (dist > 400) { return ContentService.createTextOutput(JSON.stringify({success:false, message:"Outside school: " + Math.round(dist) + "m"})) .setMimeType(ContentService.MimeType.JSON); } // Save attendance var attSheet = SpreadsheetApp.getActive().getSheetByName("Attendance_Log"); var now = new Date(); var time = Utilities.formatDate(now, Session.getScriptTimeZone(), "HH:mm:ss"); if (type === "ENTRY") { var status = "Present"; if (now.getHours() >= 10 && now.getMinutes() > 5) status = "LP"; attSheet.appendRow([now, teacher.code, teacher.name, time, "", lat, lng, Math.round(dist), status]); return ContentService.createTextOutput(JSON.stringify({success:true, message:"ENTRY - " + status})) .setMimeType(ContentService.MimeType.JSON); } if (type === "EXIT") { var data = attSheet.getDataRange().getValues(); var today = Utilities.formatDate(now, Session.getScriptTimeZone(), "dd/MM/yyyy"); for (var i = 1; i < data.length; i++) { if (data[i][0]) { var rowDate = Utilities.formatDate(new Date(data[i][0]), Session.getScriptTimeZone(), "dd/MM/yyyy"); if (rowDate === today && String(data[i][1]).trim() === teacher.code) { attSheet.getRange(i+1, 5).setValue(time); if (now.getHours() < 16) { attSheet.getRange(i+1, 9).setValue("Half Day"); return ContentService.createTextOutput(JSON.stringify({success:true, message:"EXIT - Half Day"})) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({success:true, message:"EXIT"})) .setMimeType(ContentService.MimeType.JSON); } } } return ContentService.createTextOutput(JSON.stringify({success:false, message:"Entry first"})) .setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({success:false, message:"Error"})) .setMimeType(ContentService.MimeType.JSON); } function getDistance(lat1, lon1, lat2, lon2) { var R = 6371000; var dLat = (lat2 - lat1) * Math.PI / 180; var dLon = (lon2 - lon1) * Math.PI / 180; var a = Math.sin(dLat/2) * Math.sin(dLat/2) + Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * Math.sin(dLon/2) * Math.sin(dLon/2); return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)); }
